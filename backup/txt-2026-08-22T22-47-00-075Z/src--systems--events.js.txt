function rEvent(ev){EV=ev;scene(()=>rEvent(EV));rHud();var ch='';ev.choix.forEach(function(c,i){var itemL=c.req&&!hasItem(c.req);var clsL=c.cls&&c.cls!==S.cls;var repL=(c.repMin!=null&&S.rep<c.repMin)||(c.repMax!=null&&S.rep>c.repMax);var locked=itemL||clsL||repL;var tag=clsL?(ico(CLASS_ICON[c.cls]||'dot')+' '+t(CLASSES[c.cls].n)):(repL?((c.repMin!=null)?t('ui.choice.needMin',{n:c.repMin}):t('ui.choice.needMax',{n:c.repMax})):(itemL?t('ui.choice.needItem',{name:t(OBJETS[c.req].n)}):''));ch+='<button class="btn choixbtn" '+(locked?'disabled style="opacity:.45"':'')+' onclick="choixEvent('+i+')"><b>'+(locked?ico('lock')+' ':'')+t(c.t)+'</b>'+t(c.d)+(locked&&tag?' <span style="color:#8d7d58;font-size:.8rem">'+tag+'</span>':'')+'</button>';});ec.innerHTML='<div class="ecran"><div class="parch" style="max-width:440px;margin:1.5rem auto">'+(ev.img?portrait('event',ev.img,ev.em,'nid-banner'):'')+(ev.npc?'<div style="text-align:center;margin-bottom:.4rem">'+portrait('npc',ev.npc,'🧙','npc')+'</div>':'')+'<h3>'+ev.em+' '+t(ev.titre)+'</h3><div class="orn">✦ ✦ ✦</div><p class="q">'+t(ev.txt)+'</p><div class="choix">'+ch+'</div></div></div>';}
function choixEvent(i){const __c0=EV.choix[i];if((__c0.req&&!hasItem(__c0.req))||(__c0.cls&&__c0.cls!==S.cls)||(__c0.repMin!=null&&S.rep<__c0.repMin)||(__c0.repMax!=null&&S.rep>__c0.repMax))return;sfx('page');const __ch=EV.choix[i];const res=__ch.eff();if(__ch&&__ch.consume){S.sac[__ch.consume]=Math.max(0,(S.sac[__ch.consume]||0)-1);if(S.sac[__ch.consume]<=0)delete S.sac[__ch.consume];}if(res.heroique)S.rep++;if(res.or)S.or+=res.or;EV._res=res;EV._enc=res.combat?rencontreEvent(res.combat):null;scene(viewEventResult);viewEventResult();rHud();}
function viewEventResult(){const res=EV._res;ec.innerHTML=`<div class="ecran"><div class="parch resultat" style="max-width:440px;margin:1.5rem auto">${EV.img?portrait('event',EV.img,EV.em,'nid-banner'):''}${EV.npc?`<div style="text-align:center;margin-bottom:.4rem">${portrait('npc',EV.npc,'🧙','npc')}</div>`:''}<h3>${EV.em} ${t(EV.titre)}</h3><div class="orn">✦ ✦ </div><p class="q">${t(res.txt)}</p><div style="margin-top:1rem">${res.combat?`<button class="btn btn-sang gros" onclick="startCombat(EV._enc,{})">${t('ui.face')}</button>`:`<button class="btn btn-or" onclick="retourCarte()">${t('ui.cont')}</button>`}</div></div></div>`;}
function rTresor(){scene(rTresor);rHud();ec.innerHTML=`<div class="ecran"><div class="parch" style="max-width:440px;margin:1.5rem auto"><h3>${t('ui.treasure.title')}</h3><div class="orn">✦ ✦ ✦</div><p class="q">${t('ui.treasure.txt')}</p><div class="choix"><button class="btn choixbtn" onclick="ouvrirTresor(false)"><b>${t('ui.treasure.force')}</b>${t('ui.treasure.forceD')}</button><button class="btn choixbtn" onclick="ouvrirTresor(true)"><b>${t('ui.treasure.pick')}${S.cls==='pisteuse'?t('ui.treasure.pickElf'):''}</b>${t('ui.treasure.pickD')}</button><button class="btn choixbtn" onclick="retourCarte()"><b>${t('ui.treasure.leave')}</b></button></div></div></div>`;}
function rollLootStr(){const r=Math.random();if(r<.5){const o=R(45,95);S.or+=o;return t('w.treasure.lootGold',{or:o});}else if(r<.8){const id=soinClasse();S.sac[id]=(S.sac[id]||0)+1;return t('w.treasure.lootVial',{name:t(OBJETS[id].n)});}else{const a=itemClasse(),b=itemClasse();S.sac[a]=(S.sac[a]||0)+1;S.sac[b]=(S.sac[b]||0)+1;return t('w.treasure.lootGear',{a:t(OBJETS[a].n),b:t(OBJETS[b].n)});}}
function ouvrirTresor(crochet){sfx('chest');const ambushEnc=()=>{const m=pick(['noye','charognard','goule']);return Math.random()<.4?{foes:[m,m],lead:m}:{foes:[m],lead:m};};let txt='',combat=null;if(crochet){const open=Math.random()<(S.cls==='pisteuse'?.85:.6);const ambush=Math.random()<.45;txt=open?(t('w.treasure.pickOpen')+' '+rollLootStr()):t('w.treasure.pickFail');if(ambush){txt+=' '+t('w.treasure.ambush');combat=ambushEnc();}}else{const open=Math.random()<.5;if(open){txt=t('w.treasure.forceOpen')+' '+rollLootStr();}else{S.pv=Math.max(1,S.pv-8);txt=t('w.treasure.forceTrap');}}if(!combat&&!hasItem('bague')&&!S.flags.woodcutter_done&&S.etape<=5&&Math.random()<0.3){S.sac.bague=(S.sac.bague||0)+1;txt+=' '+t('w.treasure.foundRing');}S._tresor={txt,combat};scene(viewTresorResult);viewTresorResult();rHud();}
function viewTresorResult(){const r=S._tresor||{txt:''};ec.innerHTML=`<div class="ecran"><div class="parch resultat" style="max-width:440px;margin:1.5rem auto"><h3>${t('ui.treasure.resTitle')}</h3><div class="orn">✦ ✦ ✦</div><p class="q">${r.txt}</p><div style="margin-top:1rem">${r.combat?`<button class="btn btn-sang gros" onclick="startCombat(S._tresor.combat,{})">${t('ui.face')}</button>`:`<button class="btn btn-or" onclick="retourCarte()">${t('ui.cont')}</button>`}</div></div></div>`;}
