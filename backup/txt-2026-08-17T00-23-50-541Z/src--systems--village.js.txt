function genPrimes(nd){if(nd.primesGen)return;nd.primesGen=true;nd.primesV=[];const cible=ciblePrimeDepuis(nd.id);if(cible){const elite=cible.type==='taniere';nd.primesV.push({nodeId:cible.id,m:cible.m,rec:(elite?120:70)+S.etape*5,elite,pris:false});}}
function villageNpcBlock(nd){if(nd.c!==5)return '';const done=!!S.flags.woodcutter_done;const has=hasItem('bague');let body;if(done){body=`<p class="q">${t('w.npc.woodcutter.done')}</p>`;}else{body=`<p class="q">${t('w.npc.woodcutter.txt')}</p>`+(has?`<div style="margin-top:.7rem"><button class="btn btn-or" onclick="rendreBague()">${t('w.npc.woodcutter.give')}</button></div>`:`<p class="f-fell" style="color:#7a5210;margin-top:.5rem">${t('w.npc.woodcutter.noring')}</p>`);}return `<div class="panneau section"><div style="display:flex;gap:.8rem;align-items:center;margin-bottom:.5rem">${portrait('npc','woodcutter','🪓','npc')}<h3 style="margin:0">🪓 ${t('w.npc.woodcutter.name')}</h3></div>${body}</div>`;}
function villageQuestBlock(nd){if(nd.c===0){const done=!!S.flags.pendu_decroche;let body;if(done){body=`<p class="q">${t('w.quest.pendu.done')}</p>`;}else{body=`<p class="q">${t('w.quest.pendu.txt')}</p><div style="margin-top:.7rem"><button class="btn btn-or" onclick="decrocherPendu()">${t('w.quest.pendu.btn')}</button></div>`;}return `<div class="panneau section"><h3>🪢 ${t('w.quest.pendu.name')}</h3>${body}</div>`;}if(nd.c===10){return `<div class="panneau section"><h3>📯 ${t('w.quest.baron.name')}</h3><p class="q">${t('w.quest.baron.txt')}</p></div>`;}return '';}
function decrocherPendu(){if(S.flags.pendu_decroche)return;S.flags.pendu_decroche=true;S.rep--;sfx('clic');rVillage(ND);}
function rendreBague(){if(!hasItem('bague')||S.flags.woodcutter_done)return;S.sac.bague=Math.max(0,(S.sac.bague||0)-1);if(S.sac.bague<=0)delete S.sac.bague;S.flags.woodcutter_done=true;S.rep++;S.or+=60;sfx('coin');rVillage(ND);}

function repos(){if(ND.repos)return;ND.repos=true;S.pv=ND._fullRest?S.pvMax:Math.min(S.pvMax,S.pv+Math.round(S.pvMax*.5));sfx('soin');rVillage(ND);}
function acheter(id,prix){if(S.or<prix)return;sfx('coin');S.or-=prix;S.sac[id]=(S.sac[id]||0)+1;rVillage(ND);}
function accepterPrime(i){const p=ND.primesV[i];if(!p||p.pris)return;p.pris=true;S.primes.push({nodeId:p.nodeId,m:p.m,rec:p.rec,done:false});sfx('coin');rVillage(ND);}
