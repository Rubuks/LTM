function currentActId(){return (S&&S.act)?S.act:'act1';}
function currentAct(){return (window.GameData&&window.GameData.acts&&window.GameData.acts[currentActId()])?window.GameData.acts[currentActId()]:null;}
function currentMap(){return (window.GameData&&window.GameData.maps&&window.GameData.maps[currentActId()])?window.GameData.maps[currentActId()]:null;}
function actFinalColumn(){var a=currentAct();if(a&&typeof a.finalColumn==='number')return a.finalColumn;var m=currentMap();if(m&&typeof m.finalColumn==='number')return m.finalColumn;return 11;}
function actVillageColumns(){var a=currentAct();if(a&&Array.isArray(a.villageColumns))return a.villageColumns;var m=currentMap();if(m&&Array.isArray(m.villageColumns))return m.villageColumns;return [0,5,10];}
function actLairColumns(){var a=currentAct();if(a&&Array.isArray(a.lairColumns))return a.lairColumns;var m=currentMap();if(m&&Array.isArray(m.lairColumns))return m.lairColumns;return [3,7];}
function currentMapTitleKey(){var a=currentAct();if(a&&a.mapTitleKey)return a.mapTitleKey;var m=currentMap();if(m&&m.mapTitleKey)return m.mapTitleKey;return 'ui.map.title';}
function villageNameForColumn(column){if(window.getVillageNameKey){var key=window.getVillageNameKey(currentActId(),column);if(key)return key;}var fallback={0:'w.village.v0.name',5:'w.village.v5.name',10:'w.village.v10.name'};return fallback[column]||'w.village.v0.name';}
function typeCol(c,i){var villages=actVillageColumns();var finalCol=actFinalColumn();var lairs=actLairColumns();if(villages.indexOf(c)!==-1)return 'village';if(c===finalCol)return 'nid';if(lairs.indexOf(c)!==-1&&i===0)return 'taniere';var r=Math.random();return r<.48?'combat':r<.8?'event':'tresor';}
function genMap(){const map=currentMap();const N=(map&&map.columns)?map.columns:12,nodes=[],edges=[];let id=0;const isVillage=c=>actVillageColumns().indexOf(c)!==-1;const isNid=c=>c===actFinalColumn();for(let c=0;c<N;c++){const single=isVillage(c)||isNid(c);const k=single?1:2;nodes[c]=[];for(let i=0;i<k;i++){const type=typeCol(c,i);const nd={id:id++,c,i,type,m:null,enc:null};if(type==='combat'||type==='taniere'){nd.enc=pickRencontre(c,type);nd.m=nd.enc.lead;}if(type==='nid'){const bossId=(currentAct()&&currentAct().bossId)||'griffon';nd.enc={foes:[bossId],lead:bossId};nd.m=bossId;}if(type==='village')nd.nom=villageNameForColumn(c);if(type!=='village'&&type!=='nid')nd.poi=pick(POI);nodes[c].push(nd);}}for(let c=0;c<N-1;c++){const A=nodes[c],B=nodes[c+1];if(isVillage(c+1)||isNid(c+1)){A.forEach(a=>edges.push([a.id,B[0].id]));}else if(isVillage(c)){B.forEach(b=>edges.push([A[0].id,b.id]));}else{B.forEach(b=>edges.push([pick(A).id,b.id]));A.forEach(a=>{if(!edges.some(e=>e[0]===a.id))edges.push([a.id,pick(B).id]);});}}return{nodes,edges};}
function trouveNoeud(id){for(const col of S.map.nodes)for(const nd of col)if(nd.id===id)return nd;return null;}
function nextIds(){return S.map.edges.filter(e=>e[0]===S.cur).map(e=>e[1]);}
const INNER_H=1180;
function coord(nd){const y=7+(nd.c/actFinalColumn())*85;const single=(S.map.nodes[nd.c].length===1);let x;if(single){x=50;}else{const bx=50+24*Math.sin(nd.c*0.85+0.5);x=bx+(nd.i===0?-15:15);}x=Math.max(14,Math.min(86,x));return{x,y};}
function ndLabel(nd){if(nd.type==='village')return t(nd.nom);if(nd.type==='nid')return t('ui.boss.nest');return t(nd.poi);}
function ciblePrimeDepuis(id){const file=[id],vu=new Set([id]);while(file.length){const cur=file.shift();for(const [a,b] of S.map.edges){if(a===cur&&!vu.has(b)){const nd=trouveNoeud(b);if((nd.type==='combat'||nd.type==='taniere')&&!S.visited.includes(b))return nd;vu.add(b);file.push(b);}}}return null;}
